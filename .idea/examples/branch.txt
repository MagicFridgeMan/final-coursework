def branch(x: int) -> int:
    if x > 0:
        y = x + 1
    else:
        y = x - 1
    return y

branch :: Int -> A
private l1:: Int -> A
private l2:: Int -> A
private return_expression :: Int -> A

branch(x) -> l1(x) | x > 0
l1(x) -> l2(x + 1)
l1(x) -> l3(x - 1)
l2(y) -> return_expression(y)
l3(y) -> return_expression(y)
